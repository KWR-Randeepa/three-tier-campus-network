CORE1.txt
======================================================================
!
! Packet Tracer 9.0 - Three Layer Hierarchical Network
! Device: CORE1
!
enable
configure terminal
hostname CORE1
no ip domain-lookup
vtp mode transparent
spanning-tree mode rapid-pvst
!
vlan 10
 name USERS
vlan 30
 name PRINTERS
vlan 99
 name MANAGEMENT
!
ip routing
!
interface range fastethernet 0/1-5
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,30,99
 no shutdown
!
interface vlan 10
 description USERS_GATEWAY
 ip address 192.168.10.2 255.255.255.0
 standby 10 ip 192.168.10.1
 standby 10 priority 110
 standby 10 preempt
 no shutdown
!
interface vlan 30
 description PRINTERS_GATEWAY
 ip address 192.168.30.2 255.255.255.0
 standby 30 ip 192.168.30.1
 standby 30 priority 100
 standby 30 preempt
 no shutdown
!
interface vlan 99
 description MANAGEMENT_GATEWAY
 ip address 192.168.99.2 255.255.255.0
 standby 99 ip 192.168.99.1
 standby 99 priority 110
 standby 99 preempt
 no shutdown
!
spanning-tree vlan 10,99 root primary
spanning-tree vlan 30 root secondary
!
ip dhcp excluded-address 192.168.10.1 192.168.10.20
ip dhcp excluded-address 192.168.10.101 192.168.10.254
ip dhcp pool USERS
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
 domain-name campus.local
!
end
write memory
